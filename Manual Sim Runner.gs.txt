// ManualSimRunner.gs
// Manual simulation runner — bypasses Gemini, Deka Pull, and Ebox verification.
// Use when: custom battery, unusual forklift, or any case where you already
// know the exact battery/charger combo and just need the simulation + Portée.
//
// Sheet: "Manual Sim" tab in the main spreadsheet.
// Menu:  Velox → Manual Simulation → Open Form / Run Simulation

// ─────────────────────────────────────────────────────────────────────────────
// onOpen — merged with SimulationAutomation.gs menu
// IMPORTANT: Delete the onOpen() in SimulationAutomation.gs and keep only this one.
// ─────────────────────────────────────────────────────────────────────────────
function onOpen() {
  SpreadsheetApp.getActiveSpreadsheet().addMenu('Velox', [
    { name: 'Send to Dealer',          functionName: 'sendToDealerFromRow'      },
    { name: 'Check New Emails',        functionName: 'checkForNewSimulations'   },
    { name: 'Process Queue',           functionName: 'checkPendingSimulations'  },
    null, // separator
    { name: 'Manual Simulation',       functionName: 'openManualSimSheet'         },
    { name: 'Run Manual Simulation',   functionName: 'runManualSim'               },
    { name: 'Add Working Shifts Table',functionName: 'setupWorkingShiftsTable'    },
  ]);
}

// ─────────────────────────────────────────────────────────────────────────────
// openManualSimSheet — navigates to the Manual Sim tab, creates it if missing
// ─────────────────────────────────────────────────────────────────────────────
function openManualSimSheet() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Manual Sim');
  if (!sheet) {
    sheet = _createManualSimSheet(ss);
    SpreadsheetApp.getUi().alert('✅ "Manual Sim" tab created. Fill in the fields and click Velox → Run Manual Simulation.');
  }
  ss.setActiveSheet(sheet);
}

// ─────────────────────────────────────────────────────────────────────────────
// _readManualSimParams — reads all input cells from the Manual Sim sheet
// ─────────────────────────────────────────────────────────────────────────────
function _readManualSimParams() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Manual Sim');
  if (!sheet) throw new Error('"Manual Sim" sheet not found. Run Velox → Manual Simulation first to create it.');

  function cell(row) {
    var v = sheet.getRange(row, 2).getValue();
    return v === null || v === undefined ? '' : String(v).trim();
  }
  function num(row) {
    var v = parseFloat(cell(row));
    return isNaN(v) ? 0 : v;
  }

  return {
    dealer:                cell(3),
    customer:              cell(4),
    sender_name:           cell(5),
    sender_email:          cell(6),
    language:              cell(7) || 'en',
    forklift_manufacturer: cell(9).toUpperCase(),
    forklift_model:        cell(10).toUpperCase(),
    voltage:               num(11),
    compartment:           cell(12),
    battery_ah:            num(14),
    battery_type:          cell(15) || 'VeloxPlus',
    charger_amps:          num(16),
    charger_model:         cell(17),
    battery_ah_2:          num(19),
    battery_type_2:        cell(20) || 'VeloxFlex',
    charger_amps_2:        num(21),
    charger_model_2:       cell(22),
    battery_ah_3:          num(24),
    battery_type_3:        cell(25) || 'VeloxFlex',
    charger_amps_3:        num(26),
    charger_model_3:       cell(27),
    lead_acid_ah:          num(29),
    shifts:                num(30) || 1,
    shift_duration:        num(31) || 8,
    use_type:              cell(32) || 'average',
    shift_start_hour:      cell(33) !== '' ? num(33) : 7,
    heating_battery:                     cell(34).toUpperCase() === 'YES',
    ip54_charger:                        cell(35).toUpperCase() === 'YES',
    heating_option:                      cell(36).toUpperCase() === 'YES',
    stated_charge_minutes_per_shift_raw: num(37),
    charging_pattern:                    cell(38) || 'Regular',
    notes:                               cell(40),
    gmail_thread_url:                    cell(41),
  };
}

// ─────────────────────────────────────────────────────────────────────────────
// runManualSim — main entry point called from the Velox menu
// ─────────────────────────────────────────────────────────────────────────────
function runManualSim() {
  var ui = SpreadsheetApp.getUi();
  var customBatteryLines = [];

  // ── 1. Read params from sheet ─────────────────────────────────────────────
  var p;
  try {
    p = _readManualSimParams();
  } catch(e) {
    ui.alert('❌ Error reading Manual Sim sheet:\n' + e.message);
    return;
  }

  // ── Row 36 convention: 0 = no charge during shift, 1 = auto schedule, >1 = literal minutes
  var raw36 = p.stated_charge_minutes_per_shift_raw;

  // Heated battery increases consumption from average to moderate
  if (p.heating_battery && p.use_type === 'average') {
    p.use_type = 'moderate';
  }
  p.no_charge_during_shift          = (raw36 === 0);
  p.stated_charge_minutes_per_shift = (raw36 <= 1) ? 0 : raw36;
  Logger.log('raw36 type=' + typeof raw36 + ' value=' + raw36 + ' no_charge=' + p.no_charge_during_shift);

  // ── 2. Basic validation ───────────────────────────────────────────────────
  var missing = [];
  if (!p.dealer)                missing.push('Dealer');
  if (!p.forklift_manufacturer) missing.push('Manufacturer');
  if (!p.forklift_model)        missing.push('Model');
  if (!p.voltage)               missing.push('Voltage');
  if (!p.battery_ah)            missing.push('Battery Ah (Combo 1)');
  if (!p.charger_amps)          missing.push('Charger amps (Combo 1)');
  if (!p.lead_acid_ah)          missing.push('Lead-acid Ah ref');
  if (missing.length > 0) {
    ui.alert('❌ Missing required fields:\n• ' + missing.join('\n• '));
    return;
  }
  var hasCombo2 = p.battery_ah_2 > 0 && p.charger_amps_2 > 0;
  var hasCombo3 = p.battery_ah_3 > 0 && p.charger_amps_3 > 0;

  // ── 3. Confirm ────────────────────────────────────────────────────────────
  var confirm = ui.alert(
    'Run Manual Simulation',
    p.forklift_manufacturer + ' ' + p.forklift_model + ' — ' + p.voltage + 'V\n' +
    'Combo 1: ' + p.battery_ah + 'Ah ' + p.battery_type + ' + ' + p.charger_amps + 'A\n' +
    (hasCombo2 ? 'Combo 2: ' + p.battery_ah_2 + 'Ah ' + p.battery_type_2 + ' + ' + p.charger_amps_2 + 'A\n' : '') +
    (hasCombo3 ? 'Combo 3: ' + p.battery_ah_3 + 'Ah ' + p.battery_type_3 + ' + ' + p.charger_amps_3 + 'A\n' : '') +
        (p.charging_pattern === 'Honda'
          ? 'Shifts: 2 × 9.5h (Honda schedule)\n'
          : p.charging_pattern === 'Manuel'
            ? 'Recharge: Manuel (horaire G4:L15)\n'
            : 'Shifts: ' + p.shifts + ' × ' + p.shift_duration + 'h\n') +
    'Dealer: ' + p.dealer + '\n\n' +
    'Proceed?',
    ui.ButtonSet.YES_NO
  );
  if (confirm !== ui.Button.YES) return;

  // ── 4. Build plan objects ─────────────────────────────────────────────────
  var divisor    = p.use_type === 'heavy' ? 8 : p.use_type === 'moderate' ? 9 : 10;
  var truckLoad  = Math.round(p.lead_acid_ah * 0.80 / divisor);
  var truckEff   = p.use_type === 'light' ? 50 : 80;
  var isVeloxPlus = p.battery_type.toUpperCase().includes('PLUS');

  var plan = {
    battery_ah:        p.battery_ah,
    charger_amps:      p.charger_amps,
    charger_model:     p.charger_model,
    battery_type:      p.battery_type,
    truck_load:        truckLoad,
    truck_efficiency:  truckEff,
    lead_acid_ah:      p.lead_acid_ah,
    compartment_w:     0,
    shifts:            p.shifts,
    shift_duration:    p.shift_duration,
    shift_start_hour:  p.shift_start_hour,
    voltage:           p.voltage,
    isVeloxPlusPlan:   isVeloxPlus,
    isFirstPlus:       isVeloxPlus,
  };

// ── Ebox verification (when compartment is provided) ──────────────────────
  var eboxLabel = 'CUSTOM';
  var eboxLabel2 = 'CUSTOM';
  var eboxLabel3 = 'CUSTOM';
  if (p.compartment) {
    var compParts = p.compartment.replace(/,/g, '.').replace(/×/g, 'x').split(/\s*x\s*/i);
    var compL = parseFloat(compParts[0]) || 0;
    var compW = parseFloat(compParts[1]) || 0;
    var compH = parseFloat(compParts[2]) || 0;
    if (compW > 0) {
      var eboxResult = verifyEbox(compW, p.voltage, compL, compH);
      Logger.log('Ebox verification: ' + JSON.stringify(eboxResult.fitting_eboxes));
      // Find which ebox fits combo 1
      var match1 = (eboxResult.compatible_batteries || []).find(function(b) {
        return b.ah === p.battery_ah && b.voltage === p.voltage;
      });
      var _eboxFallback = eboxResult.fitting_eboxes.length > 0 ? 'CUSTOM' : 'NO FIT';
      eboxLabel = match1 ? match1.ebox : _eboxFallback;
      // Find which ebox fits combo 2
      if (hasCombo2) {
      var match2 = (eboxResult.compatible_batteries || []).find(function(b) {
      return b.ah === p.battery_ah_2 && b.voltage === p.voltage;
      });
      eboxLabel2 = match2 ? match2.ebox : _eboxFallback;
      }
       // Find which ebox fits combo 3
       if (hasCombo3) {
      var match3 = (eboxResult.compatible_batteries || []).find(function(b) {
      return b.ah === p.battery_ah_3 && b.voltage === p.voltage;
      });
       eboxLabel3 = match3 ? match3.ebox : _eboxFallback;
         }
       }
     }

  var plan2 = null;
  if (hasCombo2) {
    var isVeloxPlus2 = p.battery_type_2.toUpperCase().includes('PLUS');
    plan2 = {
      battery_ah:        p.battery_ah_2,
      charger_amps:      p.charger_amps_2,
      charger_model:     p.charger_model_2,
      battery_type:      p.battery_type_2,
      truck_load:        truckLoad,
      truck_efficiency:  truckEff,
      lead_acid_ah:      p.lead_acid_ah,
      compartment_w:     0,
      shifts:            p.shifts,
      shift_duration:    p.shift_duration,
      shift_start_hour:  p.shift_start_hour,
      voltage:           p.voltage,
      isVeloxPlusPlan:   isVeloxPlus2,
      isFirstPlus:       false,
    };
  }

  var plan3 = null;
  if (hasCombo3) {
    var isVeloxPlus3 = p.battery_type_3.toUpperCase().includes('PLUS');
    plan3 = {
      battery_ah:        p.battery_ah_3,
      charger_amps:      p.charger_amps_3,
      charger_model:     p.charger_model_3,
      battery_type:      p.battery_type_3,
      truck_load:        truckLoad,
      truck_efficiency:  truckEff,
      lead_acid_ah:      p.lead_acid_ah,
      compartment_w:     0,
      shifts:            p.shifts,
      shift_duration:    p.shift_duration,
      shift_start_hour:  p.shift_start_hour,
      voltage:           p.voltage,
      isVeloxPlusPlan:   isVeloxPlus3,
      isFirstPlus:       false,
    };
  }

  // ── Custom battery feasibility check (after plan2/plan3 exist) ─────────────
  if ((eboxLabel === 'NO FIT' || eboxLabel === 'CUSTOM') && p.compartment) {
    var cb1 = calculateCustomBattery(p.compartment, p.voltage, p.battery_ah);
    if (cb1) {
      plan.displayBatteryType = 'VeloxCUSTOM';
      plan.customLayout = cb1;
      customBatteryLines.push('• Combo 1 — ' + p.battery_ah + 'Ah : RÉALISABLE (' + cb1.cellName + ', ' + cb1.cols + ' col × ' + cb1.rows + ' rng, ~' + cb1.achievedAh + 'Ah atteints)');
    } else {
      customBatteryLines.push('• Combo 1 — ' + p.battery_ah + 'Ah : IMPOSSIBLE (aucun arrangement ne rentre dans ce compartiment)');
    }
  }
  if (hasCombo2 && (eboxLabel2 === 'NO FIT' || eboxLabel2 === 'CUSTOM') && p.compartment) {

    var cb2 = calculateCustomBattery(p.compartment, p.voltage, p.battery_ah_2);
    if (cb2) {
      plan2.displayBatteryType = 'VeloxCUSTOM';
      plan2.customLayout = cb2;
      customBatteryLines.push('• Combo 2 — ' + p.battery_ah_2 + 'Ah : RÉALISABLE (' + cb2.cellName + ', ' + cb2.cols + ' col × ' + cb2.rows + ' rng, ~' + cb2.achievedAh + 'Ah atteints)');
    } else {
      customBatteryLines.push('• Combo 2 — ' + p.battery_ah_2 + 'Ah : IMPOSSIBLE (aucun arrangement ne rentre dans ce compartiment)');
    }
  }
  if (hasCombo3 && (eboxLabel3 === 'NO FIT' || eboxLabel3 === 'CUSTOM') && p.compartment) {
    var cb3 = calculateCustomBattery(p.compartment, p.voltage, p.battery_ah_3);
    if (cb3) {
      plan3.displayBatteryType = 'VeloxCUSTOM';
      plan3.customLayout = cb3;
      customBatteryLines.push('• Combo 3 — ' + p.battery_ah_3 + 'Ah : RÉALISABLE (' + cb3.cellName + ', ' + cb3.cols + ' col × ' + cb3.rows + ' rng, ~' + cb3.achievedAh + 'Ah atteints)');
    } else {
      customBatteryLines.push('• Combo 3 — ' + p.battery_ah_3 + 'Ah : IMPOSSIBLE (aucun arrangement ne rentre dans ce compartiment)');
    }
  }

  Logger.log('Manual sim plan: ' + JSON.stringify(plan));

  // ── 5. Build charging events ───────────────────────────────────────────────
  var chargingEvents = p.no_charge_during_shift
    ? buildPostShiftOnlyChargingEvents(p.shift_start_hour, p.shift_duration)
    : buildChargingEvents(
        p.shifts,
        p.shift_duration,
        [],
        p.stated_charge_minutes_per_shift || 0,
        p.shift_start_hour
      );

  // ── 6. Run simulations ────────────────────────────────────────────────────
  var targetEff = 100;
  var allSimResults = [];

  function runOnePlan(planObj, simNum, eboxLbl) {
    try {
      var effSearch  = findMaxEfficiency(planObj, chargingEvents, 0, targetEff);
      var maxEff     = effSearch.maxEfficiency;
      var simRes     = effSearch.finalResults;
      var effFlag    = effSearch.flag;
      var wh         = calculateWorkHours(chargingEvents, p.shifts, maxEff, p.shift_duration, p.shift_start_hour);
      var isMax      = effFlag && String(effFlag).includes('VELOXPLUS_POUSSE_AU_MAXIMUM');
      var passed     = simRes.status === 'PASS' && !isMax;
      // Re-run winning efficiency so Formula sheet recalculates correctly for screenshot
      try {
        runSingleSim(planObj, chargingEvents, maxEff);
        SpreadsheetApp.flush();
        Utilities.sleep(3000);
      } catch(e2) {}
      var soc        = [];
      try { soc = readAllSOC(); } catch(e2) {}
      try {
        var logParams = Object.assign({}, fullParams, {
          battery_ah:      planObj.battery_ah,
          charger_amps:    planObj.charger_amps,
          charger_model:   planObj.charger_model,
          battery_type:    planObj.displayBatteryType || planObj.battery_type,
          overall_passed:  passed,
          work_hours:      wh.formatted,
          truck_efficiency: maxEff,
        });
        logSimulationResult(logParams, simRes, simNum);
      } catch(e2) {}
      Logger.log('Combo ' + simNum + ': eff=' + maxEff + '% work=' + wh.formatted + ' pass=' + passed);
      return {
        passed: passed, overall_passed: passed, battery_ah: planObj.battery_ah, charger_amps: planObj.charger_amps,
        charger_model: planObj.charger_model, battery_type: planObj.battery_type,
        battery_label: (planObj.displayBatteryType || planObj.battery_type) + ' ' + p.voltage + 'V ' + planObj.battery_ah + 'Ah',
      notes: autoNotesStr,
        ebox: eboxLbl || 'CUSTOM', max_efficiency: maxEff, work_hours: wh.formatted,
        results: simRes, soc_over_time: soc, chargingEvents: chargingEvents,
        truck_load: truckLoad,
        label: 'Sim ' + simNum + ' — ' + planObj.battery_ah + 'Ah + ' + planObj.charger_amps + 'A',
        scenario: fullParams.scenario || 'MANUAL', flag: effFlag || null, simNumber: simNum, screenshotIndex: simNum - 1,
      };
    } catch(e) {
      ui.alert('❌ Simulation error (Combo ' + simNum + '):\n' + e.message);
      Logger.log('Manual sim error combo ' + simNum + ': ' + e.message);
      return null;
    }
  }

  // Need fullParams defined before runOnePlan is called — define it now
  var isMaxPushed = false; // placeholder, updated per sim below
  var overallPassed = false;
  var flags = ['MANUAL_SIM'];
  if (p.notes) flags.push(p.notes);

  // Extract Gmail thread ID from URL or direct ID for LOGS and sendToDealerFromRow
  var manualThreadId = '';
  if (p.gmail_thread_url) {
    var rawThreadVal = p.gmail_thread_url.trim();
    if (/^\d+$/.test(rawThreadVal)) {
      // Direct numeric thread ID pasted — use as-is
      manualThreadId = rawThreadVal;
      Logger.log('Thread ID from direct input: ' + manualThreadId);
    } else {
      // Try as a thread ID directly (hex format from Gmail URL)
      try {
        var t = GmailApp.getThreadById(rawThreadVal);
        if (t) { manualThreadId = t.getId(); Logger.log('Thread found by ID: ' + manualThreadId); }
      } catch(e1) {}
      // If that failed, try extracting from URL and using as thread ID
      if (!manualThreadId) {
        // Try popout URL format: th=%23thread-f%3ANUMERICID
        var popoutMatch = rawThreadVal.match(/th=%23thread-f(?:%3A|:)(\d+)/i);
        if (popoutMatch) {
          try {
            var t3 = GmailApp.getThreadById(popoutMatch[1]);
            if (t3) { manualThreadId = t3.getId(); Logger.log('Thread found by popout URL (numeric): ' + manualThreadId); }
          } catch(e2) { Logger.log('Thread ID extraction error (popout): ' + e2.message); }
          if (!manualThreadId) {
            try {
              var hexId = BigInt(popoutMatch[1]).toString(16);
              var t3b = GmailApp.getThreadById(hexId);
              if (t3b) { manualThreadId = t3b.getId(); Logger.log('Thread found by popout URL (hex): ' + manualThreadId); }
            } catch(e3) { Logger.log('Thread ID extraction error (hex): ' + e3.message); }
          }
        }
        // Try standard #inbox/HEXID format
        if (!manualThreadId) {
          var urlMatch = rawThreadVal.match(/#[^/]+\/([A-Za-z0-9]+)$/);
          if (urlMatch) {
            try {
              var t2 = GmailApp.getThreadById(urlMatch[1]);
              if (t2) { manualThreadId = t2.getId(); Logger.log('Thread found by URL fragment: ' + manualThreadId); }
            } catch(e2) { Logger.log('Thread ID extraction error: ' + e2.message); }
          }
        }
      }
      // Final fallback: search by contact email
      if (!manualThreadId && p.sender_email) {
        try {
          var searchResults = GmailApp.search('from:' + p.sender_email + ' OR to:' + p.sender_email, 0, 5);
          if (searchResults.length > 0) {
            manualThreadId = searchResults[0].getId();
            Logger.log('Thread found by contact email search: ' + manualThreadId);
          }
        } catch(e3) { Logger.log('Thread search by email error: ' + e3.message); }
      }
    }
  }

  // ── Auto-generated notes (built before fullParams so LOGS receive them) ─────
  var autoNoteLines = [];
  var eboxSummaryStr = 'Combo 1 : ' + eboxLabel;
  if (hasCombo2) {
    var lbl2 = ((eboxLabel2 === 'NO FIT' || eboxLabel2 === 'CUSTOM') && plan2 && plan2.displayBatteryType === 'VeloxCUSTOM') ? 'CUSTOM → VeloxCUSTOM' : eboxLabel2;
    eboxSummaryStr += ' | Combo 2 : ' + lbl2;
  }
  if (hasCombo3) {
    var lbl3 = ((eboxLabel3 === 'NO FIT' || eboxLabel3 === 'CUSTOM') && plan3 && plan3.displayBatteryType === 'VeloxCUSTOM') ? 'CUSTOM → VeloxCUSTOM' : eboxLabel3;
    eboxSummaryStr += ' | Combo 3 : ' + lbl3;
  }
  autoNoteLines.push('EBOX : ' + eboxSummaryStr);
  if (p.no_charge_during_shift) autoNoteLines.push('MODE : AUCUNE_CHARGE_EN_SHIFT');
  if (!p.compartment || p.compartment.trim() === '') {
    autoNoteLines.push('COMPARTIMENT : non spécifié');
    var anyNoFit = eboxLabel === 'NO FIT'
      || (hasCombo2 && eboxLabel2 === 'NO FIT')
      || (hasCombo3 && eboxLabel3 === 'NO FIT');
    if (anyNoFit) autoNoteLines.push('⚠ COMPARTIMENT MANQUANT — vérification batterie sur mesure ignorée');
  }
  if (p.ip54_charger)    autoNoteLines.push('IP54 : chargeur');
  if (p.heating_option)  autoNoteLines.push('CHAUFFAGE : chargeur');
  if (p.heating_battery) autoNoteLines.push('CHAUFFAGE : batterie (consommation → modérée)');
  if (p.lead_acid_ah)    autoNoteLines.push('ACIDE-PLOMB : ' + p.lead_acid_ah + 'Ah — USAGE : ' + p.use_type);
  if (customBatteryLines.length > 0) {
    autoNoteLines.push('--- BATTERIE SUR MESURE ---');
    customBatteryLines.forEach(function(l) { autoNoteLines.push(l); });
  }
  var autoNotesStr = autoNoteLines.join('\n');

  // ── 9. Build params object for logging / Portée ───────────────────────────
  var fullParams = {
    dealer:                p.dealer,
    customer:              p.customer,
    sender:                [p.sender_name, p.sender_email].filter(Boolean).join(' '),
    sender_name:           p.sender_name,
    sender_email:          p.sender_email,
    forklift_manufacturer: p.forklift_manufacturer,
    forklift_model:        p.forklift_model,
    voltage:               p.voltage,
    compartment:           p.compartment,
    battery_ah:            p.battery_ah,
    battery_type:          p.battery_type,
    charger_amps:          p.charger_amps,
    charger_model:         p.charger_model,
    shifts:                p.shifts,
    shift_duration:        p.shift_duration,
    use_type:              p.use_type,
    lead_acid_ah:          p.lead_acid_ah,
    truck_load:            truckLoad,
    truck_efficiency:      allSimResults.length > 0 ? allSimResults[0].max_efficiency : 0,
    work_hours:            allSimResults.length > 0 ? allSimResults[0].work_hours : '',
    language:              p.language,
    shift_start_hour:                p.shift_start_hour,
    ip54_charger:                    p.ip54_charger,
    stated_charge_minutes_per_shift: p.stated_charge_minutes_per_shift || 0,
    heating_option:        p.heating_option,
    overall_passed:        overallPassed,
    flags:                 flags,
    notes:                 autoNotesStr + (p.notes ? '\n\n' + p.notes : ''),
    scenario:              (p.stated_charge_minutes_per_shift_raw > 1) ? 'MANUAL ' + Math.round(p.stated_charge_minutes_per_shift_raw) + 'min' : 'MANUAL',
    email_subject:         'Manual Sim — ' + p.forklift_manufacturer + ' ' + p.forklift_model,
    gmail_thread_id:       manualThreadId,
  };
  // ── HONDA OVERRIDE ──────────────────────────────────────────────────────────
  if (p.charging_pattern === 'Honda') {
    fullParams.charging_pattern           = 'Honda';
    fullParams.shifts                     = 2;
    fullParams.shift_start_hour           = 6.5;
    fullParams.shift_duration             = 9.5;
    fullParams.honda_charge_per_shift_min = 62;
    plan.shifts = 2; plan.shift_duration = 9.5; plan.shift_start_hour = 6.5;
    p.shifts = 2; p.shift_duration = 9.5; p.shift_start_hour = 6.5; p.shift_duration_h = 9.5;
    if (plan2) { plan2.shifts = 2; plan2.shift_duration = 9.5; plan2.shift_start_hour = 6.5; }
    if (plan3) { plan3.shifts = 2; plan3.shift_duration = 9.5; plan3.shift_start_hour = 6.5; }
      plan.sim_start_time = '6:30 AM';
      if (plan2) { plan2.sim_start_time = '6:30 AM'; }
      if (plan3) { plan3.sim_start_time = '6:30 AM'; }
    chargingEvents = [
      { start_day: 1, start_time: '8:30 AM',  end_day: 1, end_time: '8:42 AM',  type: 'opportunity', label: 'Shift 1 Break' },
      { start_day: 1, start_time: '10:30 AM', end_day: 1, end_time: '11:00 AM', type: 'opportunity', label: 'Shift 1 Lunch' },
      { start_day: 1, start_time: '1:00 PM',  end_day: 1, end_time: '1:10 PM',  type: 'opportunity', label: 'Shift 1 Break' },
      { start_day: 1, start_time: '3:00 PM',  end_day: 1, end_time: '3:10 PM',  type: 'opportunity', label: 'Shift 1 Break' },
      { start_day: 1, start_time: '4:00 PM',  end_day: 1, end_time: '4:30 PM',  type: 'opportunity', label: 'Shift Change'  },
      { start_day: 1, start_time: '6:30 PM',  end_day: 1, end_time: '6:42 PM',  type: 'opportunity', label: 'Shift 2 Break' },
      { start_day: 1, start_time: '8:30 PM',  end_day: 1, end_time: '9:00 PM',  type: 'opportunity', label: 'Shift 2 Lunch' },
      { start_day: 1, start_time: '11:00 PM', end_day: 1, end_time: '11:10 PM', type: 'opportunity', label: 'Shift 2 Break' },
      { start_day: 2, start_time: '1:00 AM',  end_day: 2, end_time: '1:10 AM',  type: 'opportunity', label: 'Shift 2 Break' },
      { start_day: 2, start_time: '2:00 AM',  end_day: 2, end_time: '6:30 AM',  type: 'post_shift',  label: 'End of Shifts' },
    ];
  }
  // ── END HONDA OVERRIDE ──────────────────────────────────────────────────────

  // ── MANUEL OVERRIDE ──────────────────────────────────────────────────────────
  if (p.charging_pattern === 'Manuel') {
    var manuelEvents = _readManuelEvents();
    if (!manuelEvents || manuelEvents.length === 0) {
      ui.alert('❌ Mode Manuel: aucun événement de recharge trouvé dans le tableau (G4:L15).\n' +
               'Veuillez saisir au moins un événement de recharge avant de lancer la simulation.');
      return;
    }
    chargingEvents = manuelEvents;
    fullParams.charging_pattern = 'Manuel';

    // ── Derive cycle parameters from the events ──────────────────────────────
    // This ensures calculateWorkHours() and the Portée timeline both reflect
    // the real Manuel cycle duration instead of the greyed-out Shifts/Duration rows.
    var manuelCycleH    = _manuelCycleHours(manuelEvents, p.shift_start_hour);
    var manuelChargeMin = _manuelTotalChargeMin(manuelEvents);
    Logger.log('Manuel cycle: ' + manuelCycleH + 'h, total charge: ' + manuelChargeMin + ' min');

    // Embed charge minutes in scenario string so HTMLPresentationGenerator's regex
    // (/(\d+)min/i) extracts the correct value for "Charge / quart" display.
    // Must be AFTER manuelChargeMin is computed — JS var hoisting would make it undefined above.
    fullParams.scenario = 'MANUEL_' + manuelChargeMin + 'min';

    // Override p — used by calculateWorkHours (called inside runOnePlan)
    p.shifts                          = 1;
    p.shift_duration                  = manuelCycleH;
    p.stated_charge_minutes_per_shift = manuelChargeMin;

    // Derive sim_start_time string from shift_start_hour (same approach as Honda override).
    // Without this, runSingleSim writes midnight to J16 because shift_duration=24 triggers
    // a 24h-cycle code path that resets the start clock to midnight.
    var _ssh  = p.shift_start_hour;  // comes from B33 on the Manuel sim sheet
    var _sh   = Math.floor(_ssh);
    var _sm   = Math.round((_ssh - _sh) * 60);
    var _ampm = _sh < 12 ? 'AM' : 'PM';
    var _h12  = _sh % 12 || 12;
    var manuelSimStart = _h12 + ':' + (_sm < 10 ? '0' : '') + _sm + ' ' + _ampm;

    // Override plan objects — used by the Portée timeline and runSingleSim (→ J16)
    plan.shifts         = 1;
    plan.shift_duration = manuelCycleH;
    plan.sim_start_time = manuelSimStart;
    if (plan2) { plan2.shifts = 1; plan2.shift_duration = manuelCycleH; plan2.sim_start_time = manuelSimStart; }
    if (plan3) { plan3.shifts = 1; plan3.shift_duration = manuelCycleH; plan3.sim_start_time = manuelSimStart; }

    // Override fullParams — written to LOGS/SIM_DB and displayed in the Portée
    fullParams.shifts                          = 1;
    fullParams.shift_duration                  = manuelCycleH;
    fullParams.stated_charge_minutes_per_shift = manuelChargeMin;

    // ── Working Shifts table override (display/logging only) ─────────────────
    // Reads the "Quarts de travail" table (N4:R7). If filled, replaces fullParams
    // shifts/shift_duration so the LOGS/filename reflect actual work shifts
    // instead of the simulation cycle length. Does NOT affect plan or Formula sheet.
    var _manuelShifts = _readManuelShifts();
    if (_manuelShifts.length > 0) {
      fullParams.shifts         = _manuelShifts.length;
      fullParams.shift_duration = _manuelShifts[0].durationH;
    }
  }
  // ── END MANUEL OVERRIDE ──────────────────────────────────────────────────────

  // ── 10. Run simulations ───────────────────────────────────────────────────
  var sim1 = runOnePlan(plan, 1, eboxLabel);
  if (!sim1) return;
  allSimResults.push(sim1);

  if (hasCombo2) {
    var sim2 = runOnePlan(plan2, 2, eboxLabel2);
    if (sim2) allSimResults.push(sim2);
  }

  if (hasCombo3) {
    var sim3 = runOnePlan(plan3, 3, eboxLabel3);
    if (sim3) allSimResults.push(sim3);
  }


  // Determine overall status from all sims
  var anyViable = allSimResults.filter(function(s) { return s.passed; }).length;

  // ── 11. Generate Portée PDF ───────────────────────────────────────────────
  var porteePdf;
  var socArrays = allSimResults.map(function(s) { return s.soc_over_time; });
  for (var attempt = 1; attempt <= 3; attempt++) {
    try {
      porteePdf = buildHTMLPortee(fullParams, allSimResults, socArrays);
      if (porteePdf) break;
      Logger.log('buildHTMLPortee attempt ' + attempt + '/3 returned null — retrying...');
    } catch(e) {
      Logger.log('buildHTMLPortee attempt ' + attempt + '/3 error: ' + e.message);
    }
    if (attempt < 3) Utilities.sleep(15000); // wait 15s before retry
  }
  if (!porteePdf) Logger.log('buildHTMLPortee failed after 3 attempts — continuing without Portée');

  // ── 14. Capture Data Entry screenshot ────────────────────────────────────
  var dataPdf;
  try {
    Utilities.sleep(3000);
    dataPdf = capturePdf(fullParams, plan, 1, '');
  } catch(e) {
    Logger.log('capturePdf error (non-fatal): ' + e.message);
  }

  // ── 15. Send review email ─────────────────────────────────────────────────
  var allPdfs = [];
  if (porteePdf)  allPdfs.push(porteePdf);
  if (dataPdf)    allPdfs.push(dataPdf);

  // ── 15b. Generate custom battery HTML PDFs ────────────────────────────────
  var customComboPairs = [
    { plan: plan,  sim: sim1,              targetAh: p.battery_ah,   comboNum: 1 },
    { plan: plan2, sim: hasCombo2 ? sim2 : null, targetAh: p.battery_ah_2, comboNum: 2 },
    { plan: plan3, sim: hasCombo3 ? sim3 : null, targetAh: p.battery_ah_3, comboNum: 3 },

  customComboPairs.forEach(function(pair) {
    if (!pair.plan || !pair.plan.customLayout || !pair.sim) return;
    try {
      var simResult    = pair.sim;
      var workMins     = simResult.results ? (simResult.results.work_minutes || 0) : 0;
      var wh           = Math.floor(workMins / 60);
      var wm           = workMins % 60;
      var workHoursStr = wh + 'h' + (wm < 10 ? '0' : '') + wm;
      var ghostResult  = {
        layout:      pair.plan.customLayout,
        targetAh:    pair.targetAh,
        achievedAh:  pair.plan.customLayout.achievedAh,
        chargerAmps: simResult.charger_amps   || 0,
        chargerModel: simResult.charger_model || '',
        maxEff:      simResult.max_efficiency || 0,
        workHours:   workHoursStr,
        results: {
          min_soc:          simResult.results ? (simResult.results.min_soc          || 0) : 0,
          status:           simResult.passed  ? 'PASS' : 'FAIL',
          battery_used_pct: simResult.results ? (simResult.results.battery_used_pct || 0) : 0,
        },
        chargingEvents: chargingEvents,
      };
      var customHtml    = generateCustomBatteryHTML(fullParams, ghostResult);
      var customPdfName = _buildCustomBatteryPdfName(fullParams, pair.targetAh, pair.comboNum);
      var customPdfResp = UrlFetchApp.fetch(
        'https://velox-pdf-service-414761682351.us-central1.run.app/generate-pdf',
        {
          method:             'post',
          contentType:        'application/json',
          payload:            JSON.stringify({ html: customHtml, pageFormat: 'Letter' }),
          muteHttpExceptions: true,
        }
      );
      if (customPdfResp.getResponseCode() === 200) {
        var customBlob = customPdfResp.getBlob().setContentType('application/pdf');
        customBlob.setName(customPdfName);
        allPdfs.push(customBlob);
        Logger.log('Custom battery PDF generated for Combo ' + pair.comboNum + ': ' + customPdfName);

        // ── Save to Drive (dealer subfolder inside Portées) ──────────────────
        try {
          var porteeRoot   = DriveApp.getFolderById('1j1ADg0s02dKVdcCoGOZQl9IbCWt7k7Ba');
          var dealerName   = fullParams.dealer || 'Unknown';
          var dealerFolderIt = porteeRoot.getFoldersByName(dealerName);
          var dealerFolder = dealerFolderIt.hasNext()
            ? dealerFolderIt.next()
            : porteeRoot.createFolder(dealerName);
          var savedFile    = dealerFolder.createFile(customBlob.copyBlob());
          pair.customPdfId = savedFile.getId();
          Logger.log('Custom battery PDF saved to Drive: ' + pair.customPdfId);
        } catch(eDrive) {
          Logger.log('Custom battery PDF Drive save error (Combo ' + pair.comboNum + ', non-fatal): ' + eDrive.message);
        }
      } else {
        Logger.log('Custom battery PDF Cloud Run error (Combo ' + pair.comboNum + '): HTTP '
          + customPdfResp.getResponseCode() + ' — ' + customPdfResp.getContentText().substring(0, 200));
      }
    } catch(e) {
      Logger.log('Custom battery PDF error (Combo ' + pair.comboNum + ', non-fatal): ' + e.message);
    }
  });

  // ── Write to SIM_DB ────────────────────────────────────────────────────────
  try {
    var pdfId = '';
    try {
      var logsSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('LOGS');
      var lastRow   = logsSheet.getLastRow();
      var heads     = logsSheet.getRange(1, 1, 1, logsSheet.getLastColumn()).getValues()[0];
      var pdfCol    = heads.indexOf('Portée PDF ID') + 1;
      if (pdfCol > 0 && lastRow > 1) pdfId = logsSheet.getRange(lastRow, pdfCol).getValue() || '';
    } catch(e2) { Logger.log('PDF ID read error: ' + e2.message); }
    var simDbRecords = allSimResults.map(function(s) {
      return {
        battery_ah:     s.battery_ah,
        battery_type:   s.battery_type,
        charger_amps:   s.charger_amps,

  } catch(e) {

  try {

      fullParams,
      allSimResults,
      flags,
      chargingEvents,
      chargingEvents,
      allPdfs,
      0,
      60,
      false
    );

  } catch(e) {
    ui.alert('⚠️ Simulation ran but review email failed:\n' + e.message + '\n\nCheck LOGS sheet for results.');
    Logger.log('sendPipelineReviewEmail error: ' + e.message);
    return;
  }

  // ── Write custom battery PDF link to LOGS column AH ─────────────────────
  // Matches by thread ID + sim number — NOT by row position — so deduplication
  // (combo already existed → skipped) cannot cause the link to land on the
  // wrong row.
  try {
    var logsForCustom   = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('LOGS');
    var logsHeads       = logsForCustom.getRange(1, 1, 1, logsForCustom.getLastColumn()).getValues()[0];
    var threadCol       = 24; // X — Thread Gmail
    var simNumCol       = 21; // U — No. Simulation
    var customLinkCol   = 34; // AH — Custom Battery PDF Link

    // Ensure AH header is set
    if (!logsHeads[33] || logsHeads[33] === '') {
      logsForCustom.getRange(1, customLinkCol).setValue('Custom Battery PDF Link');
    }

    var hasCustoIds = customComboPairs.some(function(pair) { return !!pair.customPdfId; });
    if (hasCustoIds) {
      var currentThreadId = fullParams.gmail_thread_id || '';
      var logsLastRow     = logsForCustom.getLastRow();
      // Look back up to 100 rows — no assumption about how many were newly written
      var searchFrom  = Math.max(2, logsLastRow - 100);
      var numCols     = Math.max(logsForCustom.getLastColumn(), customLinkCol);
      var logsBlock   = logsForCustom.getRange(searchFrom, 1, logsLastRow - searchFrom + 1, numCols).getValues();
      logsBlock.forEach(function(row, idx) {
        var rowThreadId = row[threadCol - 1]; // col X (index 23)
        var rowSimNum   = row[simNumCol - 1]; // col U (index 20)
        var matchPair = customComboPairs.filter(function(pair) {
          if (!pair.customPdfId) return false;
          if (pair.comboNum != rowSimNum) return false;
          // Thread ID guard: if we have one, the row must match it exactly
          if (currentThreadId && rowThreadId != currentThreadId) return false;
          return true;
        })[0];
        if (matchPair) {
          var rowNum = searchFrom + idx;
          var pdfUrl = 'https://drive.google.com/file/d/' + matchPair.customPdfId + '/view';
          logsForCustom.getRange(rowNum, customLinkCol).setFormula(
            '=HYPERLINK("' + pdfUrl + '","📄 Batterie Sur Mesure")'
          );
          Logger.log('Custom battery PDF link written to LOGS row ' + rowNum + ' (AH)');
        }
      });
    }
  } catch(eLog) {
    Logger.log('Custom battery PDF LOGS write error (non-fatal): ' + eLog.message);
  }

  // ── Create Gmail draft for dealer ─────────────────────────────────────────
  // Draft creation disabled — use "Send to Dealer" button to send manually
  // try {
  //   _createManualSimDraft(p, allSimResults, porteePdf, pdfId, manualThreadId);
  // } catch(e) {
  //   Logger.log('Draft creation error (non-fatal): ' + e.message);
  // }

  // ── 16. Done ──────────────────────────────────────────────────────────────
  var summary = allSimResults.map(function(s) {
    return (s.passed ? '✅' : '❌') + ' Combo ' + s.simNumber + ': ' +
      s.battery_ah + 'Ah + ' + s.charger_amps + 'A — ' +
      s.max_efficiency + '% · ' + s.work_hours;
  }).join('\n');
  ui.alert(
    'Manual Simulation Complete',
    summary + '\n\nReview email sent. Check LOGS for the full entries.',
    ui.ButtonSet.OK
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// _createManualSimDraft — creates a Gmail draft from manual sim results
// Called automatically at end of runManualSim when Contact email is filled in
// ─────────────────────────────────────────────────────────────────────────────
function _createManualSimDraft(p, allSimResults, porteePdfBlob, pdfId, resolvedThreadId) {
  try {
    var toEmail   = p.sender_email;
    if (!toEmail) { Logger.log('Manual sim draft: no contact email — skipping'); return; }

    var isEN      = (p.language || 'fr').toUpperCase() === 'EN';
    var useLabel  = isEN
      ? (p.use_type === 'heavy' ? 'heavy' : p.use_type === 'moderate' ? 'moderate' : 'average')
      : (p.use_type === 'heavy' ? 'forte' : p.use_type === 'moderate' ? 'modérée' : 'moyenne');

    var shiftDesc = p.shifts + (isEN ? ' shift' + (p.shifts > 1 ? 's' : '') + ' of ' : ' quart' + (p.shifts > 1 ? 's' : '') + ' de ') + p.shift_duration + 'h';

    // Build body — one block per combo
    var body = '';
    allSimResults.forEach(function(s, idx) {
      var batType   = s.battery_ah <= 450 ? 'VeloxFlex' : s.battery_ah <= 700 ? 'VeloxPlus' : 'VeloxMax';
      var batLabel  = batType + ' ' + p.voltage + 'V ' + s.battery_ah + 'Ah';
      var workPerShift = '';
      if (s.work_hours) {
        var totalMins = _parseWorkHoursToMinutes(s.work_hours);
        var perShift  = Math.round(totalMins / (p.shifts || 1));
        var ph = Math.floor(perShift / 60);
        var pm = perShift % 60;
        workPerShift  = ph + 'h' + (pm < 10 ? '0' : '') + pm;
      }
      var chargeMin = p.stated_charge_minutes_per_shift || 60;
      var isMax     = String(s.flag || '').includes('VELOXPLUS_POUSSE_AU_MAXIMUM');

      body += 'S' + (idx + 1) + ' - ' + batLabel + ' + Omni ' + s.charger_amps + 'A';
      if (isMax) body += (isEN ? ' (pushed to maximum)' : ' (poussé au maximum)');
      body += '\n';
      if (isEN) {
        body += '  • Autonomy of ' + workPerShift + ' per shift\n';
        body += '  • ' + chargeMin + ' min of charge per shift\n';
        body += '  • ' + useLabel.charAt(0).toUpperCase() + useLabel.slice(1) + ' consumption\n\n';
      } else {
        body += '  • Autonomie de ' + workPerShift + ' par quart\n';
        body += '  • ' + chargeMin + ' minutes de charge par quart\n';
        body += '  • Consommation ' + useLabel + '\n\n';
      }
    });

    var closing = isEN
      ? 'All simulations were based on ' + useLabel + ' consumption.\n\nPlease find the Portée attached.\n\nFeel free to contact me if you have any questions.\n\nThank you!'
      : 'Toutes les simulations ont été basées sur une consommation ' + useLabel + '.\n\nVeuillez trouver la Portée en pièce jointe.\n\nN\'hésitez pas à me contacter si vous avez des questions.\n\nMerci !';
    body += closing;

    // Subject
    var shiftLabel = p.shifts + 'x' + p.shift_duration + 'h';
    var subject    = (isEN ? 'Simulation — ' : 'Simulation — ') +
                     p.forklift_manufacturer + ' ' + p.forklift_model + ' ' + p.voltage + 'V | ' + shiftLabel +
                     (p.customer ? ' | ' + p.customer : '');

    // Fetch signature
    var signature = '';
    try {
      var sendAsList   = Gmail.Users.Settings.SendAs.list('me');
      var sendAsItems  = sendAsList.sendAs || [];
      for (var si = 0; si < sendAsItems.length; si++) {
        if (sendAsItems[si].isDefault) { signature = sendAsItems[si].signature || ''; break; }
      }
    } catch(e) { Logger.log('Signature fetch error: ' + e.message); }

    var htmlBody = body.replace(/\n/g, '<br>') + '<br><br>' + signature;

    // Attach Portée PDF
    var attachments = [];
    if (pdfId) {
      try {
        var pdfFile = DriveApp.getFileById(pdfId);
        attachments.push(pdfFile.getBlob().setName(pdfFile.getName()));
      } catch(e) { Logger.log('PDF attach error: ' + e.message); }
    } else if (porteePdfBlob) {
      attachments.push(porteePdfBlob);
    }

    var draftOptions = { htmlBody: htmlBody, attachments: attachments, cc: 'quote@veloxpower.com' };

    // Thread reply vs standalone — use already-resolved thread ID
    if (resolvedThreadId) {
      try {
        var thread = GmailApp.getThreadById(resolvedThreadId);
        if (thread) {
          var messages  = thread.getMessages();
          var replyMsg  = messages[messages.length - 1];
          replyMsg.createDraftReply('', Object.assign({ to: toEmail }, draftOptions));
          Logger.log('Manual sim draft: reply created in thread for ' + toEmail);
          return;
        }
      } catch(e2) { Logger.log('Manual sim draft thread error: ' + e2.message); }
    }

    // Standalone draft
    GmailApp.createDraft(toEmail, subject, '', draftOptions);
    Logger.log('Manual sim draft: standalone draft created for ' + toEmail);

  } catch(e) {
    Logger.log('_createManualSimDraft error: ' + e.message);
  }
}

/**
 * Sanitizes a string for use as a filename segment.
 * - Removes quotes
 * - Normalises dimension separators (x, ×, X) — no spaces around them
 * - Strips all remaining spaces and special characters
 */
function _sanitizeFilePart(str) {
  return (str || '')
    .replace(/"/g, '')
    .replace(/\s*[xX×]\s*/g, 'x')
    .replace(/\s+/g, '')
    .replace(/[^a-zA-Z0-9.\-]/g, '')
    || 'NA';
}

/**
 * Builds the Drive filename for a custom battery PDF.
 * Format: CUSTOM_[Dealer]_[Client?]_[Brand]_[Model]_[Compartment]_[V]V_[Ah]Ah_Combo[N]_[YYYY-MM-DD].pdf
 */
function _buildCustomBatteryPdfName(params, targetAh, comboNum) {
  var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  var parts = [
    'CUSTOM',
    _sanitizeFilePart(params.dealer),
    params.customer ? _sanitizeFilePart(params.customer) : null,
    _sanitizeFilePart(params.forklift_manufacturer),
    _sanitizeFilePart(params.forklift_model),
    _sanitizeFilePart(params.compartment),
    (params.voltage || '') + 'V',
    (targetAh || '') + 'Ah',
    'Combo' + comboNum,
    today,
  ].filter(Boolean);
  return parts.join('_') + '.pdf';
}

// ─────────────────────────────────────────────────────────────────────────────
// _readManuelShifts — reads N4:R7 (the Working Shifts table) and returns an
// array of {startDay, startH, startTime, endDay, endH, endTime, durationH}.
// Returns [] if the table is empty or not present.
// ─────────────────────────────────────────────────────────────────────────────
function _readManuelShifts() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Manual Sim');
  if (!sheet) return [];

  // N=14, O=15, P=16, Q=17, R=18  →  rows 4–7, cols 14–18, 4 rows × 5 cols
  var data = sheet.getRange(4, 14, 4, 5).getValues();
  // data[r][0]=# (N), [1]=startDay (O), [2]=startTime (P), [3]=endDay (Q), [4]=endTime (R)

  var shifts = [];
  for (var i = 0; i < data.length; i++) {
    var row      = data[i];
    var startDay = parseInt(row[1], 10);
    var startVal = row[2];
    var endDay   = parseInt(row[3], 10);
    var endVal   = row[4];

    if (!startVal && startVal !== 0) continue;
    if (!endVal   && endVal   !== 0) continue;

    var startStr = _formatManuelTime(startVal);
    var endStr   = _formatManuelTime(endVal);
    if (!startStr || !endStr) continue;

    if (isNaN(startDay) || startDay < 1) startDay = 1;
    if (isNaN(endDay)   || endDay   < 1) endDay   = 1;
    if (startDay > 2) startDay = 2;
    if (endDay   > 2) endDay   = 2;

    var startH   = _parseTimeToHour(startStr);
    var endH     = _parseTimeToHour(endStr);
    var durationH = (endDay - startDay) * 24 + endH - startH;
    if (durationH <= 0) durationH += 24;

    shifts.push({
      startDay:  startDay,
      startH:    startH,
      startTime: startStr,
      endDay:    endDay,
      endH:      endH,
      endTime:   endStr,
      durationH: durationH,
    });
  }

  Logger.log('_readManuelShifts: parsed ' + shifts.length + ' shifts');
  return shifts;
}

// ─────────────────────────────────────────────────────────────────────────────
// _readManuelEvents — reads G4:L15 from the Manual Sim sheet and returns
// an array of {start_day, start_time, end_day, end_time, type, label} objects
// compatible with the existing chargingEvents format.
// ─────────────────────────────────────────────────────────────────────────────
function _readManuelEvents() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Manual Sim');
  if (!sheet) return [];

  // G=7, H=8, I=9, J=10, K=11, L=12 → range cols 7–12, rows 4–15
  var data = sheet.getRange(4, 7, 12, 6).getValues();
  // data[r][0]=# (G), [1]=startDay (H), [2]=startTime (I), [3]=endDay (J), [4]=endTime (K), [5]=label (L)

  var events = [];
  for (var i = 0; i < data.length; i++) {
    var row       = data[i];
    var startDay  = parseInt(row[1], 10);
    var startVal  = row[2];
    var endDay    = parseInt(row[3], 10);
    var endVal    = row[4];
    var label     = String(row[5] || '').trim();

    // Skip empty rows (no start time or no end time)
    if (!startVal && startVal !== 0) continue;
    if (!endVal   && endVal   !== 0) continue;
    var startStr = _formatManuelTime(startVal);
    var endStr   = _formatManuelTime(endVal);
    if (!startStr || !endStr) continue;

    // Default days to 1 if not filled
    if (isNaN(startDay) || startDay < 1) startDay = 1;
    if (isNaN(endDay)   || endDay   < 1) endDay   = 1;
    if (startDay > 2) startDay = 2;
    if (endDay   > 2) endDay   = 2;

    // Determine type: if it's the last non-empty event and end_day=2, treat as post_shift;
    // otherwise treat as opportunity (user-defined schedule)
    events.push({
      start_day:  startDay,
      start_time: startStr,
      end_day:    endDay,
      end_time:   endStr,
      type:       'opportunity',
      label:      label || ('Event ' + (i + 1)),
    });
  }

  // Mark the last event as post_shift if it spans to day 2 and is longer than 60 min
  // (mirrors Honda pattern where the final block is the overnight charge)
  if (events.length > 0) {
    var last = events[events.length - 1];
    if (last.end_day === 2) {
      last.type = 'post_shift';
    }
  }

  Logger.log('_readManuelEvents: parsed ' + events.length + ' events');
  return events;
}

// ─────────────────────────────────────────────────────────────────────────────
// _formatManuelTime — converts a Sheets cell value to "H:MM AM/PM" string.
// Handles Date objects (Sheets stores time-only values as Date), plain strings,
// and decimal hours (e.g. 6.5 → "6:30 AM").
// ─────────────────────────────────────────────────────────────────────────────
function _formatManuelTime(val) {
  if (!val && val !== 0) return '';

  // Date object — Sheets returns time values as Date in epoch
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'h:mm a')
      .replace('am', 'AM').replace('pm', 'PM');
  }

  var str = String(val).trim();
  if (str === '') return '';

  // Already looks like a time string — normalise AM/PM casing and return
  if (/^\d{1,2}:\d{2}\s*(AM|PM|am|pm)$/i.test(str)) {
    return str.replace(/am$/i, 'AM').replace(/pm$/i, 'PM');
  }

  // "H:MM" with no AM/PM — keep as-is (simulation engine may accept 24h)
  if (/^\d{1,2}:\d{2}$/.test(str)) {
    return str;
  }

  // Decimal hours (e.g. 6.5)
  var dec = parseFloat(str);
  if (!isNaN(dec)) {
    var h   = Math.floor(dec) % 24;
    var m   = Math.round((dec - Math.floor(dec)) * 60);
    var ampm = h < 12 ? 'AM' : 'PM';
    var h12  = h % 12 || 12;
    return h12 + ':' + (m < 10 ? '0' : '') + m + ' ' + ampm;
  }

  return str; // pass through unknown format unchanged
}

// ─────────────────────────────────────────────────────────────────────────────
// _parseTimeToHour — converts a formatted time string (e.g. "10:15 AM",
// "1:30 AM", "13:00") to a decimal hour value (0–23.99).
// Used by _manuelCycleHours and _manuelTotalChargeMin.
// ─────────────────────────────────────────────────────────────────────────────
function _parseTimeToHour(timeStr) {
  if (!timeStr) return 0;
  var s = String(timeStr).trim();

  // "H:MM AM/PM" or "HH:MM AM/PM"
  var match = s.match(/^(\d{1,2}):(\d{2})\s*(AM|PM)?$/i);
  if (match) {
    var h    = parseInt(match[1], 10);
    var m    = parseInt(match[2], 10);
    var ampm = match[3] ? match[3].toUpperCase() : null;
    if (ampm === 'PM' && h !== 12) h += 12;
    if (ampm === 'AM' && h === 12) h = 0;
    return h + m / 60;
  }

  // Decimal hours (e.g. 6.5)
  var num = parseFloat(s);
  if (!isNaN(num)) return num;
  return 0;
}

// ─────────────────────────────────────────────────────────────────────────────
// _manuelCycleHours — returns the total cycle length in hours, measured from
// shift_start_hour to the latest event end time (accounting for day 2 events).
// ─────────────────────────────────────────────────────────────────────────────
function _manuelCycleHours(events, shiftStartHour) {
  var startH   = shiftStartHour || 0;
  var maxAbsEnd = startH; // fallback: at least 1h cycle
  events.forEach(function(ev) {
    var endH   = _parseTimeToHour(ev.end_time);
    var absEnd = (ev.end_day === 2 ? 24 : 0) + endH;
    if (absEnd > maxAbsEnd) maxAbsEnd = absEnd;
  });
  return Math.max(maxAbsEnd - startH, 1);
}

// ─────────────────────────────────────────────────────────────────────────────
// _manuelTotalChargeMin — sums the duration of all Manuel charging events
// (in minutes), handling day 2 events correctly.
// ─────────────────────────────────────────────────────────────────────────────
function _manuelTotalChargeMin(events) {
  var total = 0;
  events.forEach(function(ev) {
    var startH   = _parseTimeToHour(ev.start_time);
    var endH     = _parseTimeToHour(ev.end_time);
    var absStart = (ev.start_day === 2 ? 24 : 0) + startH;
    var absEnd   = (ev.end_day   === 2 ? 24 : 0) + endH;
    var dur      = absEnd - absStart;
    if (dur > 0) total += dur * 60;
  });
  return Math.round(total);
}
