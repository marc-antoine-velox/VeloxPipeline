// ─────────────────────────────────────────────────────────────────────────────
// CLEARANCE CONSTANTS
// ─────────────────────────────────────────────────────────────────────────────

// External clearances — compartment opening → outside of Ebox (caisson)
const CAISSON_CLEARANCE_DEPTH = 0.39;   // inches removed from depth (front/back slide-in)
const CAISSON_CLEARANCE_WIDTH = 0.75;   // inches removed from width (total both sides)
// Height: no external clearance — cells never reach the top of the compartment

// Internal clearances — inside caisson wall → cells
const PACK_DIVIDER_THICKNESS  = 0.0625; // 1/16" plastic isolation sheet between packs (360°)
const CELL_WALL_CLEARANCE     = 0.125;  // min gap between cells and caisson wall — 3 sides
const BMS_SIDE_CLEARANCE      = 5.0;    // reserved for BMS + controls — 4th side only
// Total on BMS side = BMS_SIDE_CLEARANCE + CELL_WALL_CLEARANCE = 5.125"
// BMS can be on either the depth axis or width axis (technician door orientation varies by forklift)

// Vertical clearances — above cells to top of Ebox
const HEIGHT_CLEARANCE_ABOVE_CELLS = 8.37; // min vertical space above cells (wiring, terminals, lid)
const BALLAST_HEIGHT_ABOVE_CELLS   = 0;    // TODO: fill in when ballast plate height is known
// Total height consumed above cells = HEIGHT_CLEARANCE_ABOVE_CELLS + BALLAST_HEIGHT_ABOVE_CELLS

// ── Custom Ah tiers per voltage ───────────────────────────────────────────
// Each tier maps to a specific cell type via CUSTOM_CELLS validAh lists.
// Ghost sims try these in order and stop at first pass.
// standardAh = the tier used (what we quote). achievedAh = physical cell reality.
const CUSTOM_AH_TIERS_BY_VOLTAGE = {
  24:  [280, 450],
  36:  [400, 450, 550, 650],
  48:  [280, 450, 500, 550, 650, 700, 900],
  80:  [280, 450, 500, 550, 650, 700, 900, 1000],
  96:  [550, 700]
};

// ─────────────────────────────────────────────────────────────────────────────
// CELLS PER PACK LOOKUP
// Fixed table derived from Quote's layout sheet formulas.
// Math.ceil(targetAh / cellAh) is wrong for some combinations
// (e.g. CALB 163Ah @ 500Ah: ceil(500/163)=4 but correct is 3 parallel strings)
// ─────────────────────────────────────────────────────────────────────────────

const CELLS_PER_PACK = {
  'EVE_230':  { 450: 2, 700: 3, 900: 4 },
  'EVE_280':  { 280: 1, 550: 2 },
  'CALB_163': { 320: 2, 500: 3, 650: 4, 800: 5, 1000: 7 },
  'CALB_230': { 200: 1, 450: 2, 700: 3, 900: 4 }
};

function getCellsPerPack(cellKey, targetAh) {
  const table = CELLS_PER_PACK[cellKey];
  if (!table) return null;
  const count = table[targetAh];
  if (count === undefined) return null;
  return count;
}

// ─────────────────────────────────────────────────────────────────────────────
// MAIN ENTRY POINT
// Given compartment dimensions, voltage, and target Ah:
// → tries EVE cells first, then CALB
// → tries BMS on depth axis first, then width axis
// → finds best serpentine layout that fits
// → returns layout object (with feasible:true) on success
// → returns { feasible:false, attempts:[...] } with per-cell details on failure
// ─────────────────────────────────────────────────────────────────────────────

function calculateCustomBattery(compartment, voltage, targetAh) {
  const dims = parseCompartmentInput(compartment);

  if (!dims.length || !dims.width || !dims.height) {
    Logger.log('calculateCustomBattery: invalid compartment — ' + compartment);
    return null;
  }

  const compDepth  = dims.length;
  const compWidth  = dims.width;
  const compHeight = dims.height;

  const caissonDepth = compDepth - CAISSON_CLEARANCE_DEPTH;
  const caissonWidth = compWidth - CAISSON_CLEARANCE_WIDTH;

  Logger.log('calculateCustomBattery: compartment=' + compartment +
             ' | caisson=' + caissonDepth.toFixed(2) + '" deep × ' +
             caissonWidth.toFixed(2) + '" wide × ' +
             compHeight.toFixed(2) + '" tall' +
             ' | voltage=' + voltage + 'V | targetAh=' + targetAh);

  // Number of cell-packs in series for this voltage
  const seriesPacks = Math.round(voltage / 3.2);

  // BMS orientation configs — try depth axis first, then width axis.
  // BMS side reserves: CELL_WALL_CLEARANCE (gap) + BMS_SIDE_CLEARANCE (BMS + controls) + CELL_WALL_CLEARANCE (gap)
  // The other 3 sides each get CELL_WALL_CLEARANCE only.
  // Pack dividers fill the remaining space between packs (PACK_DIVIDER_THICKNESS on all sides).
  //
  // totalDepthReserved / totalWidthReserved = wall clearances + 1 leading divider on that axis
  // (mirrors the formula: available = caisson - reserved; n = floor(available / (packDim + divider)))
  var ORIENTATION_CONFIGS = [
    {
      bmsAxis:            'depth',
      // Depth axis has the BMS side + opposite wall; width axis has two plain walls
      totalDepthReserved: CELL_WALL_CLEARANCE + BMS_SIDE_CLEARANCE + CELL_WALL_CLEARANCE + PACK_DIVIDER_THICKNESS,
      totalWidthReserved: 2 * CELL_WALL_CLEARANCE + PACK_DIVIDER_THICKNESS,
      // Wall-only portions (no divider) — used in depthUsed / widthUsed formulas
      depthWallsTotal:    CELL_WALL_CLEARANCE + BMS_SIDE_CLEARANCE + CELL_WALL_CLEARANCE,
      widthWallsTotal:    2 * CELL_WALL_CLEARANCE
    },
    {
      bmsAxis:            'width',
      // Width axis has the BMS side + opposite wall; depth axis has two plain walls
      totalDepthReserved: 2 * CELL_WALL_CLEARANCE + PACK_DIVIDER_THICKNESS,
      totalWidthReserved: CELL_WALL_CLEARANCE + BMS_SIDE_CLEARANCE + CELL_WALL_CLEARANCE + PACK_DIVIDER_THICKNESS,
      depthWallsTotal:    2 * CELL_WALL_CLEARANCE,
      widthWallsTotal:    CELL_WALL_CLEARANCE + BMS_SIDE_CLEARANCE + CELL_WALL_CLEARANCE
    }
  ];

  // Try EVE first (preferred:true), then CALB
  const orderedCells = CUSTOM_CELLS.slice().sort(function(a, b) {
    if (a.preferred && !b.preferred) return -1;
    if (!a.preferred && b.preferred) return 1;
    return 0;
  });

  // Collect per-cell attempt details for failure reporting
  var attempts = [];

  for (var i = 0; i < orderedCells.length; i++) {
    const cell = orderedCells[i];

    // Start building attempt record for this cell
    var attempt = {
      cellKey:      cell.key,
      cellName:     cell.name,
      cellsPerPack: null,
      packAhIn:     null,
      packWidthIn:  null,
      packDepthIn:  null,   // pack depth (= cell depth, all cells share same depth)
      packHeightIn: null,   // pack height (= cell height, cells stand upright)
      maxCols:      null,
      maxRows:      null,
      maxPacks:     null,
      seriesNeeded: seriesPacks,
      reason:       null
    };

    // Skip if this cell type is not valid for targetAh
    if (cell.validAh.indexOf(targetAh) === -1) {
      Logger.log('  ' + cell.name + ': not valid for ' + targetAh + 'Ah — skipping');
      attempt.reason = 'Cellule non compatible avec ' + targetAh + 'Ah';
      attempts.push(attempt);
      continue;
    }

    // Cells per pack — looked up from fixed table, not calculated
    // Math.ceil is wrong for CALB 163Ah: ceil(500/163)=4 but correct answer is 3
    const cellsPerPack = getCellsPerPack(cell.key, targetAh);
    if (cellsPerPack === null) {
      Logger.log('  ' + cell.name + ': no cellsPerPack defined for ' + targetAh + 'Ah — skipping');
      attempt.reason = 'cellsPerPack non défini pour ' + targetAh + 'Ah';
      attempts.push(attempt);
      continue;
    }
    const packAh = cellsPerPack * cell.ah;

    attempt.cellsPerPack = cellsPerPack;
    attempt.packAhIn     = packAh;

    // Pack physical dimensions
    const packWidth  = cellsPerPack * cell.width;
    const packDepth  = cell.depth;
    const packHeight = cell.height;

    attempt.packWidthIn  = packWidth;
    attempt.packDepthIn  = packDepth;   // stored so HTML drawing can render the fail layout
    attempt.packHeightIn = packHeight;  // stored so front-view SVG can render cell height

    Logger.log('  ' + cell.name + ': cellsPerPack=' + cellsPerPack +
               ' packWidth=' + packWidth.toFixed(2) + '"' +
               ' packDepth=' + packDepth.toFixed(2) + '"' +
               ' packHeight=' + packHeight.toFixed(2) + '"');

    // Height check — total height consumed (cells + wiring/terminals + ballast placeholder)
    // must not exceed compartment height
    const minHeightRequired = packHeight + HEIGHT_CLEARANCE_ABOVE_CELLS + BALLAST_HEIGHT_ABOVE_CELLS;
    if (minHeightRequired > compHeight) {
      Logger.log('  ' + cell.name + ': height check failed — cells ' + packHeight.toFixed(2) +
                 '" + clearance ' + HEIGHT_CLEARANCE_ABOVE_CELLS + '" = ' +
                 minHeightRequired.toFixed(2) + '" > compartment ' + compHeight.toFixed(2) + '"');
      attempt.reason = 'Hauteur insuffisante: ' + packHeight.toFixed(2) + '" + ' +
                       HEIGHT_CLEARANCE_ABOVE_CELLS + '" (câblage/terminaux) = ' +
                       minHeightRequired.toFixed(2) + '" > ' + compHeight.toFixed(2) + '"';
      attempts.push(attempt);
      continue;
    }

    // Try both BMS orientations — depth axis first, then width axis.
    // Stop as soon as a valid layout is found.
    var bestLayout    = null;
    var bestOrientMax = 0;

    for (var oi = 0; oi < ORIENTATION_CONFIGS.length; oi++) {
      if (bestLayout) break;
      var orient = ORIENTATION_CONFIGS[oi];

      // How many packs fit along each caisson axis?
      // Formula: n × (packDim + divider) + wallsTotal + divider ≤ caissonDim
      // → n ≤ (caissonDim - totalReserved) / (packDim + divider)
      const maxPacksDeep = Math.floor((caissonDepth - orient.totalDepthReserved) / (packDepth + PACK_DIVIDER_THICKNESS));
      const maxPacksWide = Math.floor((caissonWidth - orient.totalWidthReserved) / (packWidth + PACK_DIVIDER_THICKNESS));

      // Track the best capacity seen across both orientations (for failure reporting)
      if (maxPacksDeep * maxPacksWide > bestOrientMax) {
        bestOrientMax    = maxPacksDeep * maxPacksWide;
        attempt.maxCols  = maxPacksWide;
        attempt.maxRows  = maxPacksDeep;
        attempt.maxPacks = bestOrientMax;
      }

      Logger.log('  ' + cell.name + ' [BMS côté ' + orient.bmsAxis + ']:' +
                 ' maxDeep=' + maxPacksDeep +
                 ' maxWide=' + maxPacksWide +
                 ' seriesNeeded=' + seriesPacks);

      if (maxPacksDeep === 0 || maxPacksWide === 0) {
        Logger.log('  ' + cell.name + ' [BMS côté ' + orient.bmsAxis + ']: 0 col ou rangée possible — orientation suivante');
        continue;
      }

      // Find best column count for this orientation
      // Try from max cols down to 1 — prefer more columns (cleaner serpentine)
      for (var cols = maxPacksWide; cols >= 1; cols--) {
        const rowsNeeded = Math.ceil(seriesPacks / cols);
        if (rowsNeeded > maxPacksDeep) continue;

        // Total space consumed including walls and dividers
        // depthUsed  = wall clearances (depth axis) + rows × packDepth + (rows+1) × divider
        // widthUsed  = wall clearances (width axis) + cols × packWidth + (cols+1) × divider
        const depthUsed  = orient.depthWallsTotal + rowsNeeded * packDepth  + (rowsNeeded + 1) * PACK_DIVIDER_THICKNESS;
        const widthUsed  = orient.widthWallsTotal + cols      * packWidth   + (cols      + 1) * PACK_DIVIDER_THICKNESS;
        const heightUsed = packHeight + HEIGHT_CLEARANCE_ABOVE_CELLS + BALLAST_HEIGHT_ABOVE_CELLS;

        bestLayout = {
          // Feasibility flag
          feasible:      true,

          // Cell info
          cell:          cell,
          cellKey:       cell.key,
          cellName:      cell.name,
          cellType:      cell.type,

          // Pack info
          cellsPerPack:  cellsPerPack,
          packAh:        packAh,
          packWidth:     packWidth,
          packDepth:     packDepth,
          packHeight:    packHeight,

          // Layout
          cols:          cols,
          rows:          rowsNeeded,
          totalPacks:    seriesPacks,
          totalCells:    seriesPacks * cellsPerPack,

          // Achieved battery
          achievedAh:    packAh,
          voltage:       voltage,
          targetAh:      targetAh,

          // BMS orientation used (determines which wall the door is on)
          bmsAxis:       orient.bmsAxis,

          // Dimensions used vs available
          caissonDepth:  caissonDepth,
          caissonWidth:  caissonWidth,
          compDepth:     compDepth,
          compWidth:     compWidth,
          compHeight:    compHeight,
          depthUsed:     depthUsed,
          widthUsed:     widthUsed,
          heightUsed:    heightUsed,
          depthSpare:    caissonDepth - depthUsed,
          widthSpare:    caissonWidth - widthUsed,
          // heightSpare = space above cells after wiring/terminal clearance and ballast placeholder
          // Positive = room to spare. Negative = impossible (caught by height check above).
          heightSpare:   compHeight - heightUsed,

          // Wiring path (serpentine pack numbers in series order)
          serpentinePath: buildSerpentinePath(cols, rowsNeeded, seriesPacks)
        };

        // Found a valid layout — take the first one (most columns)
        break;
      }
    }

    if (bestLayout) {
      Logger.log('calculateCustomBattery: FOUND — ' + bestLayout.cellName +
                 ' | ' + bestLayout.cols + ' cols × ' + bestLayout.rows + ' rows' +
                 ' | BMS côté ' + bestLayout.bmsAxis +
                 ' | ' + bestLayout.totalCells + ' cells (' +
                 seriesPacks + ' packs × ' + cellsPerPack + ' cells/pack)' +
                 ' | ' + bestLayout.achievedAh + 'Ah @ ' + voltage + 'V' +
                 ' | depth ' + bestLayout.depthUsed.toFixed(2) + '"/' + caissonDepth.toFixed(2) +
                 '" (spare ' + bestLayout.depthSpare.toFixed(2) + '")' +
                 ' | width ' + bestLayout.widthUsed.toFixed(2) + '"/' + caissonWidth.toFixed(2) +
                 '" (spare ' + bestLayout.widthSpare.toFixed(2) + '")' +
                 ' | height spare ' + bestLayout.heightSpare.toFixed(2) + '" au-dessus des cellules (avant poids de ballast)');
      return bestLayout;
    }

    // No layout found for this cell — record failure details
    attempt.maxCols  = attempt.maxCols  || 0;
    attempt.maxRows  = attempt.maxRows  || 0;
    attempt.maxPacks = attempt.maxPacks || 0;
    attempt.reason = 'Arrangement impossible (2 orientations BMS testées): max ' +
                     attempt.maxPacks + ' packs disponibles, ' + seriesPacks + ' requis';
    Logger.log('  ' + cell.name + ': ' + attempt.reason);
    attempts.push(attempt);
  }

  Logger.log('calculateCustomBattery: no cell fits — compartment too small for ' +
             targetAh + 'Ah at ' + voltage + 'V');
  return {
    feasible:     false,
    targetAh:     targetAh,
    voltage:      voltage,
    compartment:  compartment,
    caissonDepth: caissonDepth,
    caissonWidth:  caissonWidth,
    compHeight:   compHeight,
    seriesPacks:  seriesPacks,
    attempts:     attempts
  };
}

// ─────────────────────────────────────────────────────────────────────────────
// SERPENTINE PATH BUILDER
// Returns array of pack numbers in series wiring order.
// Even columns (0,2,4) go top→bottom (ascending row index)
// Odd columns (1,3,5) go bottom→top (descending row index)
//
// Example: 3 cols × 5 rows = 15 packs
// Grid:  col0=[1,2,3,4,5]  col1=[6,7,8,9,10]  col2=[11,12,13,14,15]
// Path:  1→2→3→4→5 (down col0)
//        → 10→9→8→7→6 (up col1, so connection 5→6 is at bottom)
//        → 11→12→13→14→15 (down col2, so connection 10→11 is at top)
// ─────────────────────────────────────────────────────────────────────────────

function buildSerpentinePath(cols, rows, totalPacks) {
  // Build grid filled in TRAVERSAL order — packs numbered by series wiring sequence
  // Even cols (0,2,4): top→bottom. Odd cols (1,3,5): bottom→top.
  var grid = [];
  var packNum = 1;

  for (var c = 0; c < cols; c++) {
    var col = new Array(rows).fill(null);
    if (c % 2 === 0) {
      // Even col: fill top→bottom
      for (var r = 0; r < rows; r++) {
        if (packNum <= totalPacks) col[r] = packNum++;
      }
    } else {
      // Odd col: fill bottom→top
      for (var r = rows - 1; r >= 0; r--) {
        if (packNum <= totalPacks) col[r] = packNum++;
      }
    }
    grid.push(col);
  }

  // Path is now simply sequential
  var path = [];
  for (var p = 1; p <= totalPacks; p++) path.push(p);

  return { path: path, grid: grid };
}

// ─────────────────────────────────────────────────────────────────────────────
// GHOST SIM RUNNER
// Runs the full simulation pipeline with a custom Ah value injected
// instead of going through Ebox lookup.
// Tries CUSTOM_AH_TIERS in order — stops at first passing tier.
// Returns { layout, simResult, chargerAmps, chargerModel } or null.
// ─────────────────────────────────────────────────────────────────────────────

function runGhostSims(params, compartment, bestMatch, eboxResult, chargers) {
  const ahTiers = CUSTOM_AH_TIERS_BY_VOLTAGE[params.voltage] ||
                  CUSTOM_AH_TIERS_BY_VOLTAGE[48];
  Logger.log('Ghost sims: trying Ah tiers for ' + params.voltage + 'V — ' + ahTiers.join(', '));

  for (var t = 0; t < ahTiers.length; t++) {
    const targetAh = ahTiers[t];

    // Check if any cell type supports this Ah tier
    const layout = calculateCustomBattery(
      compartment,
      params.voltage || bestMatch.voltage,
      targetAh
    );

    // layout is now an object on failure (feasible:false), never null unless invalid compartment
    if (!layout || layout.feasible === false) {
      Logger.log('Ghost sim ' + targetAh + 'Ah: no cell fits compartment — skipping');
      continue;
    }

    Logger.log('Ghost sim ' + targetAh + 'Ah: layout found (' + layout.cellName + ')' +
               ' — running simulation');

    // Build a ghost plan using the custom Ah
    // Use the same charger selection logic as the normal pipeline
    const voltage       = params.voltage || bestMatch.voltage;
    const useType       = params.use_type || 'average';
    const leadAcidAh    = bestMatch ? bestMatch.ah : 0;
    const divisor       = useType === 'heavy' ? 8 : useType === 'moderate' ? 9 : 10;
    const truckLoad     = leadAcidAh > 0 ? Math.round(leadAcidAh * 0.80 / divisor) : 0;
    const shiftDur      = params.shift_duration || 8;

    // Pick starting charger — same rules as selectSimulations
    const flexStartAmps = voltage === 80 ? 215 : 250;
    const ghostCharger  = chargers.find(function(c) { return c.amps >= flexStartAmps; }) ||
                          chargers[0];

    if (!ghostCharger) {
      Logger.log('Ghost sim ' + targetAh + 'Ah: no charger available — skipping');
      continue;
    }

    // Determine battery type label
    const batteryType = targetAh >= 550 ? 'VeloxCustom_Plus' : 'VeloxCustom_Flex';

    const ghostPlan = {
      battery_ah:       targetAh,
      charger_amps:     ghostCharger.amps,
      charger_model:    ghostCharger.model,
      truck_load:       truckLoad,
      truck_efficiency: 80,
      lead_acid_ah:     leadAcidAh,
      compartment_w:    bestMatch ? bestMatch.compartment_w : 0,
      shifts:           params.shifts,
      shift_duration:   shiftDur,
      battery_type:     'VeloxCustom',
      ebox:             'CUSTOM',
      conditional:      false,
      label:            'Ghost Sim — ' + targetAh + 'Ah + ' + ghostCharger.amps + 'A (VeloxCustom)'
    };

    // Build charging events
    const chargingEvents = buildChargingEvents(
      params.shifts,
      shiftDur,
      params.custom_charge_times || [],
      params.stated_charge_minutes_per_shift || 0,
      params.shift_start_hour
    );

    // Run efficiency search — same as normal pipeline
    const effSearch  = findMaxEfficiency(ghostPlan, chargingEvents, 0,
                                         params.target_efficiency || 100);
    const maxEff     = effSearch.maxEfficiency;
    const results    = effSearch.finalResults;
    const workHours  = calculateWorkHours(chargingEvents, params.shifts,
                                           maxEff, shiftDur, params.shift_start_hour);
    const workPass   = meetsMinWorkThreshold(workHours.totalMinutes, params.shifts,
                                              ghostPlan.battery_type, shiftDur);
    const passed     = results.status === 'PASS' && workPass;

    Logger.log('Ghost sim ' + targetAh + 'Ah: eff=' + maxEff + '%' +
               ' work=' + workHours.formatted +
               ' pass=' + passed);

    if (passed) {
      Logger.log('Ghost sim PASS at ' + targetAh + 'Ah — stopping tier search');
      return {
        layout:       layout,
        targetAh:     targetAh,
        achievedAh:   layout.achievedAh,
        chargerAmps:  ghostCharger.amps,
        chargerModel: ghostCharger.model,
        maxEff:       maxEff,
        workHours:    workHours.formatted,
        results:      results,
        chargingEvents: chargingEvents,
        ghostPlan:    ghostPlan
      };
    }
  }

  Logger.log('Ghost sims: no Ah tier passed — custom battery not viable for this application');
  return null;
}

function testCustomBattery_AMOD() {
  Logger.log('=== TEST: A-MOD custom battery calculator ===');

  const tiers48 = CUSTOM_AH_TIERS_BY_VOLTAGE[48];
  tiers48.forEach(function(tier) {
    const result = calculateCustomBattery('47.90 x 16.75 x 30.88', 48, tier);
    if (result && result.feasible !== false) {
      Logger.log(tier + 'Ah → ' + result.cellName +
                 ' | achieved=' + result.achievedAh + 'Ah' +
                 ' | standard=' + tier + 'Ah' +
                 ' | ' + result.cols + ' cols × ' + result.rows + ' rows' +
                 ' | BMS côté ' + result.bmsAxis +
                 ' | ' + result.totalCells + ' cells' +
                 ' | depth ' + result.depthUsed.toFixed(2) + '"/' +
                 result.caissonDepth.toFixed(2) + '"' +
                 ' | width ' + result.widthUsed.toFixed(2) + '"/' +
                 result.caissonWidth.toFixed(2) + '"' +
                 ' | height spare ' + result.heightSpare.toFixed(2) + '"' +
                 ' | fits=' + (result.depthUsed <= result.caissonDepth &&
                                result.widthUsed <= result.caissonWidth ? 'YES' : 'NO'));
      if (tier === 450) {
        Logger.log('  Serpentine path: ' + result.serpentinePath.path.join(' → '));
      }
    } else {
      var reasons = (result && result.attempts) ?
        result.attempts.map(function(a) { return a.cellName + ': ' + a.reason; }).join('; ') :
        'compartiment invalide';
      Logger.log(tier + 'Ah → NO FIT for A-MOD compartment — ' + reasons);
    }
  });

  Logger.log('=== TEST COMPLETE ===');
}

  const compartment = '32.68 x 29.06 x 24.69';
